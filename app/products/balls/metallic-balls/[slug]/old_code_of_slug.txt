import Link from "next/link";
import { notFound } from "next/navigation";
import { FiChevronRight, FiArrowRight } from "react-icons/fi";
import { getBallProductDetail, ballProductDetails } from "@/lib/data/ball-product-details";
import CrushingLoadsSection from "@/components/CrushingLoadsSection";

export function generateStaticParams() {
  return Object.keys(ballProductDetails).map((slug) => ({ slug }));
}

// ─────────────────────────────────────────────
// SPEC TABLE — element/property + value, always left-aligned
// ─────────────────────────────────────────────
function SpecTable({
  title,
  rows,
}: {
  title: string;
  rows: { label: string; value: string }[];
}) {
  return (
    <div>
      <h2 className="text-sm font-semibold uppercase tracking-widest text-orange-500 mb-3 text-center">
        {title}
      </h2>
      <div className="rounded-xl border border-gray-100 overflow-hidden">
        {rows.map((row, i) => (
          <div
            key={row.label}
            className={`px-4 py-2.5 text-sm flex flex-col gap-1 sm:flex-row sm:items-start sm:gap-4 ${i % 2 === 0 ? "bg-white" : "bg-gray-50"
              }`}
          >
            <span className="text-gray-500 shrink-0 sm:w-32">{row.label}</span>
            <span className="text-gray-800 text-left leading-relaxed">{row.value}</span>
          </div>
        ))}
      </div>
    </div>
  );
}

// ─────────────────────────────────────────────
// GRADE TABLE — element/property + one column per grade
// Header is center-aligned, matching SpecTable
// Used for products WITH gradeColumns (420/420C, 302/304/304L, 316/316L, Brass)
// ─────────────────────────────────────────────
function GradeTable({
  title,
  gradeColumns,
  rows,
}: {
  title: string;
  gradeColumns: string[];
  rows: { label: string; values: string[] }[];
}) {
  return (
    <div>
      <h2 className="text-sm font-semibold uppercase tracking-widest text-orange-500 mb-3 text-center">
        {title}
      </h2>
      <div className="rounded-xl border border-gray-100 overflow-hidden">
        <table className="w-full text-sm">
          <thead className="bg-orange-50">
            <tr>
              <th className="px-4 py-2 text-left"></th>
              {gradeColumns.map((g) => (
                <th key={g} className="px-4 py-2 text-left">
                  {g}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {rows.map((row, i) => (
              <tr key={row.label} className={i % 2 === 0 ? "bg-white" : "bg-gray-50"}>
                <td className="px-4 py-2.5 text-gray-500">{row.label}</td>
                {row.values.map((v, j) => (
                  <td key={j} className="px-4 py-2.5 text-gray-800">
                    {v}
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

// ─────────────────────────────────────────────
// PAGE COMPONENT
// ─────────────────────────────────────────────
export default async function BallProductDetailPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const product = getBallProductDetail(slug);

  if (!product) {
    notFound();
  }

  return (
    <main className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-10">
      {/* ── Breadcrumb ── */}
      <nav className="flex flex-wrap items-center gap-1.5 text-sm text-gray-400 mb-8">
        <Link href="/" className="hover:text-orange-500 transition-colors no-underline">
          Home
        </Link>
        <FiChevronRight size={13} />
        <Link href="/products" className="hover:text-orange-500 transition-colors no-underline">
          Products
        </Link>
        <FiChevronRight size={13} />
        <Link href="/products/balls" className="hover:text-orange-500 transition-colors no-underline">
          Balls
        </Link>
        <FiChevronRight size={13} />
        <Link
          href="/products/balls/metallic-balls"
          className="hover:text-orange-500 transition-colors no-underline"
        >
          Metallic Balls
        </Link>
        <FiChevronRight size={13} />
        <span className="text-gray-800 font-medium">{product.name}</span>
      </nav>

      {/* ══════════════════════════════════
          HERO — Title + Quick facts (no image, no download button)
      ══════════════════════════════════ */}
      <section className="flex flex-col rounded-2xl overflow-hidden border border-gray-100 shadow-sm bg-white mb-10">
        <div className="flex flex-col justify-center w-full px-6 py-7 sm:px-10 sm:py-10">
          <span className="text-xs font-semibold uppercase tracking-widest text-orange-500 mb-3">
            {product.material}
          </span>
          <h1 className="text-2xl sm:text-3xl font-bold text-gray-900 leading-tight mb-4">
            {product.name}
          </h1>
          <p className="text-gray-500 text-sm leading-relaxed mb-6">{product.generalInfo}</p>

          {(product.sizeRange || product.standardGrades) && (
            <div className="flex flex-wrap items-center  gap-6 pt-5 border-t border-gray-100">
              {product.sizeRange && (
                <div>
                  <p className="text-xs text-gray-400 mt-0.5 text-left">Size Range</p>
                  <p className="text-xs font-bold text-orange-500 text-left">{product.sizeRange}</p>

                </div>
              )}
              {product.standardGrades && (
                <div>
                  <p className="text-xs text-gray-400 mt-0.5 text-left">Standard Grades</p>
                  <p className="text-xs font-bold text-orange-500 text-left">{product.standardGrades}</p>

                </div>
              )}
            </div>
          )}
        </div>
      </section>

      {/* ══════════════════════════════════
          INTERNATIONAL EQUIVALENTS — only if present
      ══════════════════════════════════ */}
      {product.internationalEquivalents && (
        <section className="mb-10">
          <h2 className="text-sm font-semibold uppercase tracking-widest text-orange-500 mb-3">
            International Equivalents
          </h2>
          <p className="text-gray-600 text-sm leading-relaxed bg-gray-50 rounded-xl px-5 py-4">
            {product.internationalEquivalents}
          </p>
        </section>
      )}

      {/* ══════════════════════════════════
          APPLICATIONS — only if present
      ══════════════════════════════════ */}
      {product.applications && (
        <section className="mb-10">
          <h2 className="text-sm font-semibold uppercase tracking-widest text-orange-500 mb-3">
            Applications
          </h2>
          <p className="text-gray-600 text-sm leading-relaxed bg-gray-50 rounded-xl px-5 py-4">
            {product.applications}
          </p>
        </section>
      )}

      {/* ══════════════════════════════════
          CHEMICAL COMPOSITION + MECHANICAL PROPERTIES + HARDNESS
          Branches per product:
          - gradeColumns present  -> GradeTable (multi-column: 420/420C, 302/304/304L, 316/316L, Brass)
          - gradeColumns absent   -> SpecTable (single-value, everything else)
      ══════════════════════════════════ */}
      <section
        className={`grid gap-8 mb-10 ${product.hardness
          ? "grid-cols-1 lg:grid-cols-3"
          : "grid-cols-1 sm:grid-cols-2"
          }`}
      >
        {product.gradeColumns ? (
          <GradeTable
            title="Chemical Composition"
            gradeColumns={product.gradeColumns}
            rows={product.chemicalComposition.map((c) => ({
              label: c.element,
              values: c.values ?? [c.value ?? ""],
            }))}
          />
        ) : (
          <SpecTable
            title="Chemical Composition"
            rows={product.chemicalComposition.map((c) => ({
              label: c.element,
              value: c.value ?? "",
            }))}
          />
        )}

        {product.mechanicalProperties && (
          <SpecTable
            title="Mechanical Properties"
            rows={product.mechanicalProperties.map((m) => ({
              label: m.property,
              value: m.value,
            }))}
          />
        )}

        {product.hardness && (
          product.gradeColumns && product.hardness.some((h) => h.values) ? (
            <GradeTable
              title="Hardness"
              gradeColumns={product.gradeColumns}
              rows={product.hardness.map((h) => ({
                label: h.key,
                values: h.values ?? [h.value ?? ""],
              }))}
            />
          ) : (
            <SpecTable
              title="Hardness"
              rows={product.hardness.map((h) => ({
                label: h.key,
                value: h.value ?? "",
              }))}
            />
          )
        )}
      </section>

      {/* ══════════════════════════════════
          MINIMUM CASE DEPTH
      ══════════════════════════════════ */}
      {product.minimumCaseDepth && (
        <section className="mb-10">
          <h2 className="text-sm font-semibold uppercase tracking-widest text-orange-500 mb-3">
            Minimum Case Depth
          </h2>
          <div className="overflow-hidden rounded-xl border border-gray-100">
            <table className="w-full text-sm">
              <thead className="bg-orange-50">
                <tr>
                  <th className="px-4 py-3 text-left">Ball Diameter</th>
                  <th className="px-4 py-3 text-left">Case Depth</th>
                </tr>
              </thead>
              <tbody>
                {product.minimumCaseDepth.map((row, i) => (
                  <tr key={i} className={i % 2 === 0 ? "bg-white" : "bg-gray-50"}>
                    <td className="px-4 py-3">{row.ballDia}</td>
                    <td className="px-4 py-3">{row.caseDepth}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </section>
      )}

      {/* ══════════════════════════════════
          CRUSHING LOADS — split into 2 side-by-side tables.
          Adapts columns based on whether rows carry
          diaInch+diaMm (Chrome Alloy) or just ballDia (Through Hardened).
      ══════════════════════════════════ */}
      {product.crushingLoads && (
        <CrushingLoadsSection rows={product.crushingLoads} />
      )}

      {/* ══════════════════════════════════
          BACK LINK
      ══════════════════════════════════ */}
      <div className="flex justify-center">
        <Link
          href="/products/balls/metallic-balls"
          className="flex items-center gap-2 text-sm font-semibold text-orange-500 hover:text-orange-600 transition-colors no-underline"
        >
          <FiArrowRight size={14} className="rotate-180" />
          Back to Metallic Balls
        </Link>
      </div>
    </main>
  );
}
